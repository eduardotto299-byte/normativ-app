    <script>

        const standards = {

            'ОП': [{number:'10',five:'15 сек',four:'17 сек',three:'19 сек'},{number:'11',five:'25 сек',four:'27 сек',three:'32 сек'},{number:'13',five:'33 сек',four:'38 сек',three:'43 сек'}],

            'РХБЗ': [{number:'1',five:'7 сек',four:'8 сек',three:'10 сек'},{number:'4',five:'3:00',four:'3:20',three:'4:00'},{number:'7',five:'3:20',four:'3:40',three:'4:30'},{number:'8',five:'35 сек',four:'40 сек',three:'50 сек'},{number:'13',five:'8 сек',four:'10 сек',three:'12 сек'}],

            'ТСБП': [{number:'27',five:'50 сек',four:'55 сек',three:'60 сек'},{number:'28',five:'5 сек',four:'4 сек',three:'3 сек'},{number:'32',five:'50 сек',four:'55 сек',three:'60 сек'},{number:'33',five:'1:15',four:'1:25',three:'1:35'}],

            'ПС': [{number:'1',five:'2:30',four:'3:30',three:'4:30'}],

            'ИП': [{number:'1',five:'25 сек',four:'27 сек',three:'32 сек'},{number:'6',five:'15 сек',four:'17 сек',three:'21 сек'}],

            'ВТ': [{number:'1',five:'40 сек',four:'45 сек',three:'55 сек'},{number:'3',five:'45 сек',four:'50 сек',three:'1:00'}],

            'ВМП': [{number:'1',five:'1:10',four:'1:15',three:'1:30'},{number:'2',five:'18 сек',four:'20 сек',three:'25 сек'}]

        };

        const fpExercises = [

            {number:'4',unit:'раз',thresholds:{'до 25':{five:'14',four:'12',three:'10'},'25-30':{five:'13',four:'11',three:'9'},'30-35':{five:'12',four:'10',three:'8'},'35-40':{five:'10',four:'8',three:'6'},'40-45':{five:'8',four:'6',three:'4'}}},

            {number:'51',unit:'сек',thresholds:{'до 25':{five:'14,0',four:'14,4',three:'15,2'},'25-30':{five:'14,2',four:'14,6',three:'15,6'},'30-35':{five:'14,8',four:'15,4',three:'16,2'},'35-40':{five:'15,2',four:'15,8',three:'16,8'},'40-45':{five:'16,0',four:'16,6',three:'17,6'}}},

            {number:'52',unit:'сек',thresholds:{'до 25':{five:'26,5',four:'27,5',three:'28,5'},'25-30':{five:'27,5',four:'28,5',three:'29,5'},'30-35':{five:'28,5',four:'29,5',three:'30,5'},'35-40':{five:'29,5',four:'30,5',three:'31,5'},'40-45':{five:'31,0',four:'32,0',three:'34,0'}}},

            {number:'55',unit:'мин',thresholds:{'до 25':{five:'3:20',four:'3:30',three:'3:55'},'25-30':{five:'3:25',four:'3:35',three:'4:05'},'30-35':{five:'3:30',four:'3:40',three:'4:15'},'35-40':{five:'3:45',four:'4:00',three:'4:35'},'40-45':{five:'3:55',four:'4:10',three:'4:40'}}},

            {number:'56',unit:'мин',thresholds:{'до 25':{five:'12:10',four:'12:25',three:'13:00'},'25-30':{five:'12:20',four:'12:35',three:'13:10'},'30-35':{five:'12:30',four:'12:45',three:'13:20'},'35-40':{five:'14:00',four:'14:45',three:'15:30'},'40-45':{five:'-',four:'-',three:'-'}}}

        ];

        const ageGroups = ['до 25', '25-30', '30-35', '35-40', '40-45'];

        let targetGrade = 4, peopleCount = 1, selectedNorms = {}, selectedFpNorms = {}, selectedFpAge = 'до 25';

        const storage = { get: (k) => { try { return JSON.parse(localStorage.getItem(k)); } catch(e) { return null; } }, set: (k, v) => { try { localStorage.setItem(k, JSON.stringify(v)); } catch(e) {} } };

 

        function init() { renderSubjects(); setupTabs(); setupControls(); loadState(); }

        function renderSubjects() {

            const container = document.getElementById('subjects-container');

            let html = '';

            for (const [subj, norms] of Object.entries(standards)) {

                html += `<div class="expandable" data-subj="${subj}"><div class="card-title"><span>${subj}</span><span class="chevron">▼</span></div><div class="expandable-content">${norms.map(n => `<label class="checkbox-item"><input type="checkbox" data-norm="${subj}_${n.number}"><span>Норматив №${n.number}</span></label>`).join('')}</div></div>`;

            }

            html += `<div class="expandable" data-subj="ФП"><div class="card-title"><span>🏃 ФП (Физподготовка)</span><span class="chevron">▼</span></div><div class="expandable-content"><div style="margin-bottom: 12px;"><div style="font-size: 13px; margin-bottom: 8px; font-weight: 500;">Возрастная группа:</div><div class="segmented-control" id="age-control">${ageGroups.map((age, i) => `<button type="button" class="segment ${i===0?'active':''}" data-age="${age}">${age}</button>`).join('')}</div></div>${fpExercises.map(ex => `<label class="checkbox-item"><input type="checkbox" data-fp="${ex.number}"><span>№${ex.number} (${ex.unit})</span></label>`).join('')}</div></div>`;

            container.innerHTML = html;

            document.querySelectorAll('.expandable').forEach(el => {

                el.addEventListener('click', function(e) {

                    if (e.target.tagName === 'INPUT' || e.target.tagName === 'LABEL') return;

                    this.querySelector('.expandable-content').classList.toggle('show');

                    this.querySelector('.chevron').classList.toggle('rotated');

                });

            });

        }

        function setupTabs() {

            document.querySelectorAll('.tab').forEach(tab => {

                tab.addEventListener('click', () => {

                    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));

                    document.querySelectorAll('.content').forEach(c => c.classList.remove('active'));

                    tab.classList.add('active');

                    document.getElementById(`tab-${tab.dataset.tab}`).classList.add('active');

                });

            });

        }

        function setupControls() {

            document.querySelectorAll('#grade-control .segment').forEach(btn => {

                btn.addEventListener('click', () => {

                    document.querySelectorAll('#grade-control .segment').forEach(b => b.classList.remove('active'));

                    btn.classList.add('active');

                    targetGrade = parseInt(btn.dataset.grade);

                });

            });

            document.querySelectorAll('#age-control .segment').forEach(btn => {

                btn.addEventListener('click', () => {

                    document.querySelectorAll('#age-control .segment').forEach(b => b.classList.remove('active'));

                    btn.classList.add('active');

                    selectedFpAge = btn.dataset.age;

                    storage.set('fpAge', selectedFpAge);

                });

            });

            document.getElementById('dec-people').addEventListener('click', () => { if(peopleCount>1){peopleCount--; document.getElementById('people-count').textContent=peopleCount;}});

            document.getElementById('inc-people').addEventListener('click', () => { peopleCount++; document.getElementById('people-count').textContent=peopleCount;});

            document.getElementById('calc-plan').addEventListener('click', calculatePlan);

            document.getElementById('clear-plan').addEventListener('click', () => { document.getElementById('plan-results').innerHTML=''; document.getElementById('clear-plan').style.display='none';});

            document.getElementById('calc-progress').addEventListener('click', calculateProgress);

            document.getElementById('clear-progress').addEventListener('click', () => { ['planned-count','actual-count','count-5','count-4','count-3'].forEach(id => document.getElementById(id).value=''); document.getElementById('progress-results').innerHTML=''; document.getElementById('clear-progress').style.display='none';});

            document.addEventListener('change', e => {

                if (e.target.dataset.norm) { e.target.checked ? selectedNorms[e.target.dataset.norm] = true : delete selectedNorms[e.target.dataset.norm]; }

                if (e.target.dataset.fp) { e.target.checked ? selectedFpNorms[e.target.dataset.fp] = true : delete selectedFpNorms[e.target.dataset.fp]; }

                storage.set('norms', {selectedNorms, selectedFpNorms});

            });

        }

        function calculatePlan() {

            const resDiv = document.getElementById('plan-results');

            if (Object.keys(selectedNorms).length === 0 && Object.keys(selectedFpNorms).length === 0) { alert('Выберите хотя бы один норматив!'); return; }

            let html = '<div class="card" style="margin-top:16px;"><div class="card-title">📋 Результаты</div>';

            for (let p = 1; p <= peopleCount; p++) {

                html += `<div style="margin-bottom:16px; padding-bottom:16px; border-bottom:1px solid var(--border);"><div style="font-weight:700; margin-bottom:8px;">👤 Человек #${p} (Общая: ${targetGrade})</div>`;

                for (const key of Object.keys(selectedNorms)) {

                    const [subj, num] = key.split('_');

                    const data = standards[subj].find(n => n.number === num);

                    const val = data[targetGrade===5?'five':targetGrade===4?'four':'three'];

                    html += `<div class="result-card"><div class="result-header"><span class="result-title">${subj} №${num}</span><span class="result-grade">${targetGrade}</span></div><div class="result-value">${val}</div></div>`;

                }

                for (const num of Object.keys(selectedFpNorms)) {

                    const ex = fpExercises.find(e => e.number === num);

                    const th = ex.thresholds[selectedFpAge];

                    const val = targetGrade===5?th.five:targetGrade===4?th.four:th.three;

                    html += `<div class="result-card"><div class="result-header"><span class="result-title">ФП №${num}</span><span class="result-grade">${targetGrade}</span></div><div class="result-value">${val} ${ex.unit}</div></div>`;

                }

                html += '</div>';

            }

            html += '</div>';

            resDiv.innerHTML = html;

            document.getElementById('clear-plan').style.display = 'block';

            resDiv.scrollIntoView({behavior:'smooth', block:'start'});

        }

        function calculateProgress() {

            const planned = parseInt(document.getElementById('planned-count').value) || 0;

            const actual = parseInt(document.getElementById('actual-count').value) || 0;

            const c5 = parseInt(document.getElementById('count-5').value) || 0;

            const c4 = parseInt(document.getElementById('count-4').value) || 0;

            const c3 = parseInt(document.getElementById('count-3').value) || 0;

            if (!planned || !actual) { alert('Введите план и факт!'); return; }

            const total = c5 + c4 + c3;

            const execPercent = Math.min(100, Math.round((actual/planned)*100));

            const avg = total ? ((c5*5 + c4*4 + c3*3)/total).toFixed(2) : '0.00';

            let html = `<div class="card" style="margin-top:16px;"><div style="text-align:center; padding:20px; background:#f8f9f5; border-radius:12px; margin-bottom:16px;"><div style="font-size:14px; color:var(--text-sec); margin-bottom:8px;">Выполнение плана</div><div style="font-size:42px; font-weight:700; color:var(--primary);">${execPercent}%</div><div class="progress-bar"><div class="progress-fill" style="width:${execPercent}%"></div></div><div style="font-size:13px; color:var(--text-sec);">из ${planned} выполнили ${actual}</div></div><div style="margin-bottom:16px;"><div style="font-weight:600; margin-bottom:12px; font-size:14px;">Распределение оценок:</div>${c5?`<div class="stat-item"><span class="stat-label">Оценка 5</span><span class="stat-value" style="color:var(--primary);">${c5} (${Math.round(c5/total*100)}%)</span></div>`:''}${c4?`<div class="stat-item"><span class="stat-label">Оценка 4</span><span class="stat-value" style="color:#7a9947;">${c4} (${Math.round(c4/total*100)}%)</span></div>`:''}${c3?`<div class="stat-item"><span class="stat-label">Оценка 3</span><span class="stat-value" style="color:#888;">${c3} (${Math.round(c3/total*100)}%)</span></div>`:''}</div><div style="padding:16px; background:#f8f8f8; border-radius:10px; text-align:center;"><div style="font-size:13px; color:var(--text-sec);">Средний балл</div><div style="font-size:28px; font-weight:700; color:var(--primary);">${avg}</div></div></div>`;

            document.getElementById('progress-results').innerHTML = html;

            document.getElementById('clear-progress').style.display = 'block';

            document.getElementById('progress-results').scrollIntoView({behavior:'smooth', block:'start'});

        }

        function loadState() {

            const saved = storage.get('norms');

            if (saved) {

                selectedNorms = saved.selectedNorms || {};

                selectedFpNorms = saved.selectedFpNorms || {};

                document.querySelectorAll('input[data-norm]').forEach(cb => { cb.checked = !!selectedNorms[cb.dataset.norm]; });

                document.querySelectorAll('input[data-fp]').forEach(cb => { cb.checked = !!selectedFpNorms[cb.dataset.fp]; });

            }

            const savedAge = storage.get('fpAge');

            if (savedAge) {

                selectedFpAge = savedAge;

                document.querySelectorAll('#age-control .segment').forEach(b => { b.classList.toggle('active', b.dataset.age === savedAge); });

            }

        }

        window.addEventListener('DOMContentLoaded', init);

    </script>
